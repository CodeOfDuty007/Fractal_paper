\node[gpuphase, minimum width=30mm, text width=38mm] (att) at (2.0,3.5)

   (\texttt{human\_revision} excluded)};

\node[gpuphase, minimum width=26mm, text width=32mm] (resc) at (6.9,1.2)

\node[phase, minimum width=26mm, text width=32mm] (fam) at (6.9,3.5)

\node[artifact, minimum width=28mm] (v2) at (6.9,5.9)

% ---------------------------------------------------------------- battery
% moved right from x=12.0: at text width 70mm the box spanned x 8.5-15.5 and

\node[evalnode, inner sep=6pt] (bat) at (13.4,4.4)
  {\begin{tabular}{@{}l r@{}}
     \multicolumn{2}{@{}l}{\textbf{Robustness battery} --- 5 detectors $\times$ 6 attacks}\\
     \multicolumn{2}{@{}l}{\scriptsize recall@1\%FPR, clean $\rightarrow$ worst attack}\\[3pt]
     \texttt{roberta}           & $0.9989 \rightarrow 0.9959$\\
     \texttt{binoculars}        & $0.8728 \rightarrow 0.7456$\\
     \texttt{tabular} \scriptsize(85-feat) & $0.9559 \rightarrow \textcolor{okVermil}{\mathbf{0.2670}}$\\
     \texttt{fastdetectgpt}     & $0.5104 \rightarrow 0.1967$\\
     \texttt{perplexity\_gemma} & $0.0935$\ \emph{\scriptsize(below floor)}\\
   \end{tabular}};

\node[finding, minimum width=58mm, text width=72mm] (pat) at (13.4,0.9)
  {\textbf{Three distinct failure modes.}
   (i) \texttt{tabular} collapses \emph{specifically} on sentence-reorder ---
   an order-sensitive detector.
   (ii) \texttt{fastdetectgpt} recall \emph{rises} under paraphrase: a
   same-family artifact (Gemma-paraphrased text is maximally predictable to
   the Gemma scorer); it reverses under a non-lineage attacker.
   (iii) \texttt{roberta} is near-unbreakable across all six.};
